// TinyMind 自动生成的固件代码
// 目标：STM32F407 | 模型：KWS | 量化：INT8

#include "stm32f4xx_hal.h"
#include "tinymind_model.h"
#include "audio_capture.h"

#define SAMPLE_RATE   16000
#define FRAME_SIZE    480
#define MODEL_INPUT   490

static int8_t audio_buffer[FRAME_SIZE];
static int8_t model_output[12];

int main(void) {
    // 系统初始化
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_USART1_UART_Init();
    MX_ADC1_Init();

    // TinyMind模型初始化
    tinymind_model_init();
    audio_capture_init(SAMPLE_RATE);

    printf("TinyMind KWS Ready!\r\n");

    while (1) {
        // 采集音频
        audio_capture_read(audio_buffer, FRAME_SIZE);

        // 模型推理
        tinymind_invoke(audio_buffer, model_output);

        // 检测关键词
        if (model_output[0] > 0.8f) {
            HAL_GPIO_TogglePin(GPIOC, GPIO_PIN_13);
            printf("Keyword detected!\r\n");
        }
    }
}
