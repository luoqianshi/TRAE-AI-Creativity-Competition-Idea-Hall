核心技术：本地 Canvas 压缩算法

智能二分压缩，在清晰度与文件大小间找到最佳平衡

// utils/image.js - 核心压缩算法

async function compressToTargetSize({ sourcePath, cropRect, spec, context }) {
  const outputFormat = 'jpg';
  let bestUnderMax = null;
  let quality = 1;
  const minQuality = 0.35;
  const step = 0.06;

  // 从高质量逐步降低，寻找第一个符合大小的图片
  while (quality >= minQuality) {
    const outputPath = await renderToTargetSize({
      sourcePath,
      cropRect,
      targetWidth: spec.width,
      targetHeight: spec.height,
      quality,
      context
    });

    const info = await getPhotoInfo(outputPath);

    if (info.sizeKB <= spec.maxSizeKB) {
      const result = buildProcessResult(outputPath, spec, info, outputFormat, quality);
      
      // 如果同时满足最小大小要求，直接返回
      if (!spec.minSizeKB || info.sizeKB >= spec.minSizeKB) {
        return result;
      }

      // 否则记录当前最佳结果继续尝试
      if (!bestUnderMax || info.sizeKB > bestUnderMax.outputSizeKB) {
        bestUnderMax = result;
      }
    }

    quality = Number((quality - step).toFixed(2));
  }

  // 如果找到过符合最大大小的结果，返回最优的那一个
  if (bestUnderMax) return bestUnderMax;

  return {
    success: false,
    errorCode: 'COMPRESS_FAILED',
    errorMessage: '这张照片无法在保证清晰度的情况下压缩到指定大小，请换一张照片再试。'
  };
}
算法特点：从 quality=1 开始逐步下降，优先保证清晰度；如果无法满足最小大小要求，返回最接近且不超过最大大小的结果；绝不低于 0.35 质量阈值，避免生成极度模糊的照片。

核心原则与设计理念

✓ 纯前端实现

无后端服务、无云存储、无登录、无数据库，所有逻辑在小程序端完成
