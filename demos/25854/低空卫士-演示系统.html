电子围栏入侵检测
// 计算目标到围栏边界距离
public checkFenceIntrusion(Target target) {
  for (Fence fence : enabledFences) {
    double distance =
      calculateDistance(target, fence);
    if (distance <= fence.getThreshold()) {
      createAlert(target, fence,
        AlertLevel.fromDistance(distance));
      pushAlert(alert); // WebSocket推送
    }
  }
}

// TargetSimulator定时任务
@Scheduled(fixedRate = 1000)
public void simulateTargetMovement() {
  for (Target t : simulatedTargets) {
    // 直线/圆周/巡逻/悬停模式
    Position nextPos = t.calculateNextPosition();
    t.setPosition(nextPos);
    saveTrajectory(t, nextPos);
    broadcast("/ws/target", t);
    checkFenceIntrusion(t);
  }
}
