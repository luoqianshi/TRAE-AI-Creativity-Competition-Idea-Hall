Rust 代码示例

// Cargo.toml
[dependencies]
idiom-mnemonic = "0.1"  // 四字成语助记词库

// 核心依赖
sha2 = "0.10"
hmac = "0.12"
pbkdf2 = "0.12"

// 熵处理
rand = "0.8"
hex = "0.4"
use idiom_mnemonic::{MnemonicGenerator, MnemonicType};

// 生成 12 个四字成语助记词 (128 bit 熵)
fn generate_12_words() -> Result<Vec<String>, Error> {
    let entropy = generate_secure_entropy(16);  // 128 bit
    
    let mnemonic = MnemonicGenerator::new(MnemonicType::12)
        .with_entropy(&entropy)?
        .generate()?;
    
    let words: Vec<String> = mnemonic
        .into_words()
        .iter()
        .map(|w| w.to_string())
        .collect();
    
    println("助记词: {:?}", words);
    Ok(words)
}

// 生成 24 个四字成语助记词 (256 bit 熵)
fn generate_24_words() -> Result<Vec<String>, Error> {
    let entropy = generate_secure_entropy(32);  // 256 bit
    
    MnemonicGenerator::new(MnemonicType::24)
        .with_entropy(&entropy)?
        .generate()
        .map(|m| m.into_words())
}
use idiom_mnemonic::{MnemonicGenerator, MnemonicType};
use sha2::{Sha256, Digest};

// 从助记词导出 BIP32 种子
fn mnemonic_to_seed(mnemonic: &[String], passphrase: &str) 
    -> Vec<u8> 
{
    // 1. 助记词拼接 + normalization
    let phrase = mnemonic.join(" ");
    let normalized = normalize(&phrase);  // BIP39 normalization
    
    // 2. PBKDF2 拉伸 (2048 轮 HMAC-SHA512)
    let salt = format("mnemonic{}", passphrase);
    let seed = pbkdf2_sha512(normalized.as_bytes(), salt.as_bytes());
    
    // 3. 返回 64 字节种子 (BIP32 HDKD)
    seed[0..64].to_vec()
}

// 验证助记词完整性
fn verify_checksum(mnemonic: &[String]) -> bool {
    let generator = MnemonicGenerator::new(MnemonicType::12);
    generator.verify(mnemonic)
}
