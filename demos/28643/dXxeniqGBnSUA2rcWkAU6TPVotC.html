import requests, base64

# 火山引擎方舟 API 端点
API_URL = "https://ark.cn-beijing.volces.com/api/v3/chat/completions"

def call_vl_api(frame_path, round_data):
    img_b64 = base64.b64encode(open(frame_path, "rb").read()).decode()
    payload = {
        "model": "doubao-1.5-vision-pro-32k",
        "messages": [{
            "role": "user",
            "content": [
                {"type": "text", "text": VL_PROMPT + str(round_data)},
                {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{img_b64}"}}
            ]
        }]
    }
    resp = requests.post(API_URL, headers=headers, json=payload)
    return resp.json()

# 解析K系数，修正隐性贡献
res = call_vl_api(slice_img, current_round_info)
for item in res["player_effect"]:
    pid, k = item["player_id"], item["k_value"]
    if "掩护" in item["action"]:
        player_off_hidden[pid] *= k
    if "延误" in item["action"]:
        player_def_hidden[pid] *= k
