    <script>
        let money = 100;
        let currentPrice = 0;
        let canScratch = false;
        let winPrize = 0;

        const moneyEl = document.getElementById('money');
        const resultEl = document.getElementById('result');
        const pandaEl = document.getElementById('panda');
        const tipEl = document.getElementById('tipText');
        const maskCanvas = document.getElementById('maskCanvas');
        const ctx = maskCanvas.getContext('2d');
        const btns = document.querySelectorAll('.btn[data-price]');
        const resetBtn = document.getElementById('resetBtn');

        const pandaNormal = 'https://p.qqan.com/up/2022-7/1658644531739424.jpg';
        const pandaWin = 'https://pic.qqtn.com/up/2022-7/1658644542751124.jpg';
        const pandaLose = 'https://pic.qqtn.com/up/2022-7/1658644547756444.jpg';

        // 画布尺寸
        function resizeCanvas() {
            const box = maskCanvas.parentElement.getBoundingClientRect();
            maskCanvas.width = box.width;
            maskCanvas.height = box.height;
            ctx.fillStyle = '#999';
            ctx.fillRect(0, 0, maskCanvas.width, maskCanvas.height);
        }
        resizeCanvas();
        window.addEventListener('resize', resizeCanvas);

        // 刷新余额按钮状态
        function updateUI() {
            moneyEl.textContent = money;
            btns.forEach(btn => {
                const price = parseInt(btn.dataset.price);
                btn.disabled = money < price;
            });
        }

        // 随机开奖
        function startLottery(price) {
            if (money < price) return;
            currentPrice = price;
            money -= price;
            updateUI();

            // 概率设置：60%不中奖，30%小奖，9%中，1%大奖
            const rand = Math.random();
            if (rand < 0.6) {
                winPrize = 0;
                resultEl.textContent = '谢谢参与';
                pandaEl.src = pandaLose;
                tipEl.textContent = '手气一般，下次一定中！';
                document.getElementById('loseAudio').currentTime = 0;
                document.getElementById('loseAudio').play();
            } else if (rand < 0.9) {
                winPrize = Math.floor(price * (1 + Math.random() * 2));
                money += winPrize;
                resultEl.textContent = `恭喜中奖 ${winPrize} 元`;
                pandaEl.src = pandaWin;
                tipEl.textContent = `小幸运！赚了${winPrize}元`;
                document.getElementById('winAudio').currentTime = 0;
                document.getElementById('winAudio').play();
            } else if (rand < 0.99) {
                winPrize = price * 5;
                money += winPrize;
                resultEl.textContent = `中二等奖 ${winPrize} 元`;
                pandaEl.src = pandaWin;
                tipEl.textContent = '运气不错，小暴富！';
                document.getElementById('winAudio').currentTime = 0;
                document.getElementById('winAudio').play();
            } else {
                winPrize = 100;
                money += winPrize;
                resultEl.textContent = `超级大奖 ${winPrize} 元！`;
                pandaEl.src = pandaWin;
                tipEl.textContent = '天选熊猫人，直接暴富！';
                document.getElementById('winAudio').currentTime = 0;
                document.getElementById('winAudio').play();
            }

            updateUI();
            // 重置刮奖涂层
            ctx.fillStyle = '#999';
            ctx.fillRect(0, 0, maskCanvas.width, maskCanvas.height);
            canScratch = true;
        }

        // 刮奖逻辑
        let isDrawing = false;
        function scratch(x, y) {
            if (!canScratch) return;
            document.getElementById('scratchAudio').currentTime = 0;
            document.getElementById('scratchAudio').play();
            ctx.globalCompositeOperation = 'destination-out';
            ctx.beginPath();
            ctx.arc(x, y, 25, 0, Math.PI * 2);
            ctx.fill();
        }

        maskCanvas.addEventListener('mousedown', e => {
            isDrawing = true;
            const rect = maskCanvas.getBoundingClientRect();
            scratch(e.clientX - rect.left, e.clientY - rect.top);
        });
        maskCanvas.addEventListener('mousemove', e => {
            if (!isDrawing) return;
            const rect = maskCanvas.getBoundingClientRect();
            scratch(e.clientX - rect.left, e.clientY - rect.top);
        });
        maskCanvas.addEventListener('mouseup', () => isDrawing = false);
        maskCanvas.addEventListener('mouseleave', () => isDrawing = false);

        // 按钮绑定
        btns.forEach(btn => {
            btn.onclick = () => {
                const p = parseInt(btn.dataset.price);
                startLottery(p);
            };
        });

        // 重置资金
        resetBtn.onclick = () => {
            money = 100;
            canScratch = false;
            currentPrice = 0;
            winPrize = 0;
            resultEl.textContent = '请先选择刮奖面额';
            pandaEl.src = pandaNormal;
            tipEl.textContent = '来试试手气，看看熊猫人能不能暴富！';
            ctx.fillStyle = '#999';
            ctx.fillRect(0, 0, maskCanvas.width, maskCanvas.height);
            updateUI();
        };

        updateUI();
    </script>
