// 核心判断：协议握手能否完成
function handshake(phone, charger, cable) {
  if (!charger.pd) return { ok:false, phase:'PHASE 01', cause:'充电头不支持 PD' };
  if (!cable.pd)  return { ok:false, phase:'PHASE 02', cause:'线材缺 E-Marker 芯片' };
  if (cable.max < charger.out) return { ok:true, throttle:true, cause:'线材限流降速' };
  return { ok:true, power: Math.min(phone.maxPD, charger.out, cable.max) };
}
