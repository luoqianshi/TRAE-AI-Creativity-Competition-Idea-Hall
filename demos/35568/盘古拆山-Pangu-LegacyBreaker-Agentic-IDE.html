modern_auth_service.py (重构为微服务) 安全 & 规范 

from fastapi import FastAPI, Depends, HTTPException, status
from sqlalchemy.orm import Session
from schemas import UserLoginResponse, ShopResponse
from security import verify_password
from database import get_db
import models

app = FastAPI(title="Modern Auth Service")

@app.post("/api/v2/auth/login", response_model=UserLoginResponse)
async def login(usr: str, pwd: str, user_type: int = 1, db: Session = Depends(get_db)):
    """
    用户/商户统一登录接口
    重构说明：使用 FastAPI + SQLAlchemy 替代原生 PHP + 裸写 SQL。
    规避了原版的 SQL 注入风险，并使用 bcrypt 替代明文密码。
    """
    # 使用 ORM 防止 SQL 注入
    user = db.query(models.User).filter(models.User.name == usr).first()
    
    if not user or not verify_password(pwd, user.password_hash):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Incorrect username or password"
        )
        
    if user_type == 1:
        if user.status != 1:
            raise HTTPException(status_code=403, detail="User account is locked")
        return {"code": 200, "msg": "success", "data": user}
        
    elif user_type == 2:
        shop = db.query(models.Shop).filter(models.Shop.uid == user.id).first()
        if not shop:
            raise HTTPException(status_code=404, detail="Shop not found")
        return {"code": 200, "shop_data": shop}
        
    else:
        raise HTTPException(status_code=400, detail="Invalid user_type parameter")
