function 匹配教程(设备信息, 问题分类) {
  const 品牌 = 设备信息.brand || 用户手动选择;
  const 系统 = 识别系统版本(设备信息.userAgent);
  const 候选教程 = 教程库.filter(item =>
    item.适配品牌.includes(品牌) &&
    item.分类 === 问题分类
  );
  return 按照匹配度排序(候选教程)[0] ||
    返回通用安卓/苹果教程;
}
