    function switchPipeline(fileObj){
        var is3899 = fileObj.version === '3.8.99';
        var targetPipeline = is3899 ? 2 : 1;

        if(targetPipeline === _currentPipeline){
            if(_currentPipeline === 1) { disposePlayer(); createPlayer(fileObj); }
            else { disposeMode2(); m2LoadFile(fileObj); }
            return;
        }

        if(_currentPipeline === 1) disposePlayer();
        else disposeMode2();

        _currentPipeline = targetPipeline;

        if(targetPipeline === 1){
            showMode1UI();
            createPlayer(fileObj);
        } else {
            showMode2UI();
            ensureMode2Runtime(function(){ m2LoadFile(fileObj); });
        }
    }

    // ===== 降级模式（无服务器 / 直接打开 HTML 时使用 .gif 预览）=====

    // 降级模式 .gif 文件列表
    // - 服务器模式：通过 /api/gif-list 自动扫描，无需手动维护

    var _gifFiles = ['pet.gif']; /* GIF_FILES_AUTO_INJECT */

        // file:// 协议 = 直接打开 HTML，无服务器

        return false;
    }

function m2LoadFile(fileObj){
        var dir = 'spine-3.8.99';
        var baseName = fileObj.name;
        var ext = fileObj.ext; // 'json' or 'skel'

        setM2Status('[3.8.99] 加载 ' + baseName + '.' + ext + ' ...', '');

        // 1) 并行加载 atlas 文本 + 骨骼数据
        var atlasPromise = fetch(dir + '/' + baseName + '.atlas').then(function(r){ return r.text(); });
        var dataPromise = ext === 'json'
            ? fetch(dir + '/' + baseName + '.json').then(function(r){ return r.json(); })
            : fetch(dir + '/' + baseName + '.skel').then(function(r){ return r.arrayBuffer(); });

        // 2) atlas 加载完成后：解析页图名称 → 全部预加载+decode → 再创建 WebGL 资源
        atlasPromise.then(function(atlasText){
            // ---- 解析 atlas 页图名称 ----
            // Spine atlas 格式: 页图名(独立行) → size/format/filter/pma 行 → 空行 → region块...
            var lines = atlasText.split(/\r?\n/);
            var pageNames = [];
            for(var li = 0; li < lines.length; li++){
                var ln = lines[li].trim();
                if(!ln) continue;
                // 跳过所有已知属性行
                if(ln.startsWith('size:') || ln.startsWith('format:') ||
                   ln.startsWith('filter:') || ln.startsWith('repeat') ||
                   ln.startsWith('pma') || ln.startsWith('rotate') ||
                   ln.startsWith('coords') || ln.startsWith('indices') ||
                   ln.startsWith('end')) continue;
                // 页图名特征: 独立行，且前一行是空行（或这是第一行）
                if(li === 0 || !lines[li-1].trim()){
                    // 进一步确认: 不包含等号、不全是数字（排除 region 的 bounds/offsets 等）
                    if(!ln.includes('=') && !ln.includes(':') && isNaN(Number(ln))){
                        pageNames.push(ln);
                    }
                }
            }

            console.log('[mode2] atlas pages:', pageNames);

            if(!pageNames.length){
                setM2Status('[3.8.99] 未找到页图', 'err');
                return;
            }

            // 3) 预加载全部页图 — 每张图都必须 onload + decode 完成才继续
            var imgMap = {}; // { pageName: HTMLImageElement }
            var preloadPromises = pageNames.map(function(pname){
                return new Promise(function(resolve){
                    var img = new Image();
                    img.crossOrigin = 'anonymous';
                    img.onload = function(){
                        // 关键: decode() 确保 GPU 就绪后再 resolve
                        if(typeof img.decode === 'function'){
                            img.decode().then(function(){
                                imgMap[pname] = img;
                                resolve();
                            }).catch(function(){
                                // decode 失败也接受（某些浏览器不支持）
                                imgMap[pname] = img;
                                resolve();
                            });
                        } else {
                            imgMap[pname] = img;
                            resolve();
                        }
                    };
                    img.onerror = function(){
                        console.error('[mode2] 页图加载失败:', pname);
                        resolve(); // 不阻塞其他图片
                    };
                    img.src = dir + '/' + pname;
                });
            });

            Promise.all([dataPromise].concat(preloadPromises)).then(function(allResults){
                var rawData = allResults[0];

                try {
                    // 4) 所有资源就绪后初始化 WebGL
                    m2Canvas = document.getElementById('myCanvas');
                    m2Canvas.width = m2Canvas.offsetWidth || 1200;
                    m2Canvas.height = 960;

                    var context = new spineWebgl.webgl.ManagedWebGLRenderingContext(m2Canvas, { alpha:true, premultipliedAlpha:true });
                    m2GL = context;
                    m2Shader = spineWebgl.webgl.Shader.newTwoColoredTextured(context);
                    m2Batcher = new spineWebgl.webgl.PolygonBatcher(context);
                    m2Renderer = new spineWebgl.webgl.SkeletonRenderer(context);
                    m2Renderer.premultipliedAlpha = document.getElementById('m2-pma').checked; // 跟随复选框状态
                    m2MVP = new spineWebgl.webgl.Matrix4();

                    // 5) 创建 TextureAtlas — 回调中直接从 imgMap 取已加载完成的图片
                    var textureAtlas = new spineWebgl.TextureAtlas(atlasText, function(pageName){
                        var img = imgMap[pageName];
                        if(!img){
                            console.error('[mode2] 页图未预加载:', pageName, '可用:', Object.keys(imgMap));
                            throw new Error('页图缺失: ' + pageName);
                        }
                        return new spineWebgl.webgl.GLTexture(context, img);
                    });

                    var attachmentLoader = new spineWebgl.AtlasAttachmentLoader(textureAtlas);

                    // 解析骨骼数据
                    var skeletonData;
                    if(ext === 'skel'){
                        var bin = new spineWebgl.SkeletonBinary(attachmentLoader);
                        bin.scale = 1;
                        skeletonData = bin.readSkeletonData(new Uint8Array(rawData));
                    } else {
                        var jsonParser = new spineWebgl.SkeletonJson(attachmentLoader);
                        jsonParser.scale = 1;
                        skeletonData = jsonParser.readSkeletonData(rawData);
                    }

                    m2SkeletonData = skeletonData;

                    // 创建 Skeleton + AnimationState
                    m2Skeleton = new spineWebgl.Skeleton(skeletonData);
                    var stateData = new spineWebgl.AnimationStateData(skeletonData);
                    if(typeof stateData.setDefaultMix === 'function') stateData.setDefaultMix(0.2);
                    else stateData.defaultMix = 0.2;
                    m2State = new spineWebgl.AnimationState(stateData);

                    // 填充皮肤下拉
                    var skinsSel = document.getElementById('m2-skins');
                    skinsSel.innerHTML = '';
                    if(skeletonData.skins){
                        var defaultSkin = skeletonData.defaultSkin ? skeletonData.defaultSkin.name : 'default';
                        skeletonData.skins.forEach(function(sk){
                            var opt = document.createElement('option');
                            opt.value = sk.name; opt.textContent = sk.name;
                            if(sk.name === defaultSkin) opt.selected = true;
                            skinsSel.appendChild(opt);
                        });
                    }
                    skinsSel.onchange = function(){
                        if(!m2Skeleton) return;
                        var skinName = skinsSel.value;
                        m2Skeleton.setToSetupPose();
                        m2Skeleton.setSkinByName(skinName);
                        m2Skeleton.setSlotsToSetupPose();
                        m2UpdateViewport();
                    };

                    // 初始化时显式设置默认皮肤（否则部分资源的默认皮肤attachment不会绑定到slot）
                    if(skinsSel.value){
                        m2Skeleton.setToSetupPose();
                        m2Skeleton.setSkinByName(skinsSel.value);
                        m2Skeleton.setSlotsToSetupPose();
                    }

                    // 填充动画下拉
                    var animSel = document.getElementById('m2-animations');
                    animSel.innerHTML = '';
                    if(skeletonData.animations){
                        skeletonData.animations.forEach(function(a){
                            var opt = document.createElement('option');
                            opt.value = a.name; opt.textContent = a.name;
                            animSel.appendChild(opt);
                        });
                    }
                    animSel.onchange = function(){
                        var name = animSel.value;
                        if(name && m2State){
                            m2State.setAnimation(0, name, true);
                            m2UpdateViewport();
                        }
                    };

                    // ---- Track 2 表情轨道 ---- //
                    var track2Wrap = document.getElementById('m2-track2-wrap');
                    var track2Sel  = document.getElementById('m2-track2-select');
                    if(track2Wrap && track2Sel && skeletonData.animations && skeletonData.animations.length > 0){
                        track2Sel.innerHTML = '<option value="">-- 无 --</option>';
                        skeletonData.animations.forEach(function(a){
                            var opt = document.createElement('option');
                            opt.value = a.name; opt.textContent = a.name;
                            track2Sel.appendChild(opt);
                        });
                        track2Wrap.style.display = '';
                        track2Sel.onchange = function(){
                            var name = this.value;
                            if(!name || !m2State){ return; }
                            try {
                                m2State.setAnimation(1, name, true);
                                console.log('[m2 track2] setAnimation(1):', name);
                            } catch(e){ console.warn('[m2 track2] 播放失败:', e.message); }
                        };
                    } else if(track2Wrap){ track2Wrap.style.display = 'none'; }

                    // 播放第一个动画
                    if(skeletonData.animations && skeletonData.animations.length > 0){
                        var firstAnim = skeletonData.animations[0].name;
                        m2State.setAnimation(0, firstAnim, true);
                        animSel.value = firstAnim;
                    }
                    m2State.update(0);
                    m2State.apply(m2Skeleton);
                    m2Skeleton.updateWorldTransform();
                    m2UpdateViewport();

                    // 状态信息
                    setM2Status('[3.8.99] ' + baseName + ' 加载完成', 'ok');
                    var infoEl = document.getElementById('m2-info');
                    infoEl.innerHTML =
                        '版本 <span>3.8.99</span>' +
                        ' &nbsp;|&nbsp; 文件 <span>' + baseName + '</span>' +
                        ' &nbsp;|&nbsp; 格式 <span>.' + ext + '</span>';
                    if(skeletonData.animations)
                        infoEl.innerHTML += ' &nbsp;|&nbsp; 动画 <span>' + skeletonData.animations.length + '</span> 个';

                    // 启动渲染循环
                    m2Playing = true;
                    var playBtn = document.getElementById('m2-play-btn');
                    if(playBtn) playBtn.textContent = '暂停';
                    m2Loop();

                    // 窗口缩放时同步 canvas 分辨率，防止变形
                    window._m2ResizeHandler = function(){
                        if(!m2Canvas || _currentPipeline !== 2) return;
                        var newW = m2Canvas.offsetWidth || 1200;
                        var newH = m2Canvas.offsetHeight || 960;
                        if(m2Canvas.width !== newW || m2Canvas.height !== newH){
                            m2Canvas.width = newW;
                            m2Canvas.height = newH;
                            m2UpdateViewport();
                        }
                    };
                    window.addEventListener('resize', window._m2ResizeHandler);
                    // 立即执行一次，确保初始尺寸正确
                    window._m2ResizeHandler();

                } catch(err){
                    console.error('[mode2] load error:', err);
                    setM2Status('[3.8.99] 解析失败: ' + err.message, 'err');
                }
            }).catch(function(err){
                console.error('[mode2] resource error:', err);
                setM2Status('[3.8.99] 资源加载失败: ' + err.message, 'err');
            });
        }).catch(function(err){
            setM2Status('[3.8.99] Atlas 加载失败: ' + err.message, 'err');
        });
    }

    function m2UpdateViewport(){
        if(!m2Skeleton || !m2Canvas) return;
        var minX=-300,minY=-300,maxX=300,maxY=300;
        try {
            m2Skeleton.updateWorldTransform();
            var offset = new spineWebgl.Vector2();
            var size = new spineWebgl.Vector2();
            m2Skeleton.getBounds(offset, size, []);
            if(isFinite(offset.x)&&isFinite(offset.y)&&isFinite(size.x)&&isFinite(size.y)&&size.x>1&&size.y>1){
                minX=offset.x; minY=offset.y; maxX=offset.x+size.x; maxY=offset.y+size.y;
            }
        } catch(e){}
        var w=Math.max(1,maxX-minX), h=Math.max(1,maxY-minY), pad=0.72;
        m2ViewScale = Math.min(m2Canvas.width/w, m2Canvas.height/h)*pad;
        if(!isFinite(m2ViewScale)||m2ViewScale<=0) m2ViewScale=1;
        var cx=minX+w/2, cy=minY+h/2;
        m2ViewX=m2Canvas.width/2-cx*m2ViewScale;
        m2ViewY=m2Canvas.height/2+cy*m2ViewScale;
    }

    function m2Loop(){
        if(_currentPipeline!==2||!m2Playing||!m2Skeleton||!m2State||!m2Renderer) return;
        m2AnimId = requestAnimationFrame(m2Loop);
        var now = Date.now();
        var delta=(now-(m2LastTime||now))/1000;
        m2LastTime = now;
        if(delta>0.1) delta=0.1;

        m2State.update(delta);
        m2State.apply(m2Skeleton);
        m2Skeleton.updateWorldTransform();

        var gl = m2GL.gl;
        gl.viewport(0,0,m2Canvas.width,m2Canvas.height);
        gl.clearColor(0,0,0,0);
        gl.clear(gl.COLOR_BUFFER_BIT);
        gl.enable(gl.BLEND);
        gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA);

        var worldW=m2Canvas.width/m2ViewScale, worldH=m2Canvas.height/m2ViewScale;
        var worldX=-m2ViewX/m2ViewScale, worldY=(m2ViewY-m2Canvas.height)/m2ViewScale;
        m2MVP.ortho2d(worldX,worldY,worldW,worldH);

        m2Shader.bind();
        m2Shader.setUniformi(spineWebgl.webgl.Shader.SAMPLER, 0);
        m2Shader.setUniform4x4f(spineWebgl.webgl.Shader.MVP_MATRIX, m2MVP.values);
        m2Batcher.begin(m2Shader);
        m2Renderer.draw(m2Batcher, m2Skeleton);
        m2Batcher.end();
        m2Shader.unbind();
    }

    var _wheelTarget = null;

    // ---- CSS transform 缁勫悎 ----  //
    var _zoomScale = 1;

    // ---- 婊氳疆缂╂斁 ----  //
    // 鐢ㄧ嫭绔嬪彉閲忎繚瀛?wheel 鐩爣锛屼笉涓?drag 鐨?_wheelTarget 娣风敤
    var _zoomStageEl = null;

    function bindWheelZoom(player){
        unbindWheelZoom();
        var stage = document.getElementById('zoom-stage');
        if(!stage) return;
        _zoomStageEl = stage;
        _wheelHandler = function(e){
            e.preventDefault();
            e.stopPropagation();
            var delta = e.deltaY || e.detail || 0;
            var factor = 1 - delta * 0.0006;   // 姣?00px deltaY 绾︾缉鏀?%
            if(factor < 0.88) factor = 0.88;   // 鍗曟鏈€澶х缉灏?2%
            if(factor > 1.14) factor = 1.14;   // 鍗曟鏈€澶ф斁澶?4%
            _zoomScale *= factor;
            if(_zoomScale < ZOOM_MIN) _zoomScale = ZOOM_MIN;
            if(_zoomScale > ZOOM_MAX) _zoomScale = ZOOM_MAX;
            updateStageTransform();
            var zoomPct = Math.round(_zoomScale * 100);
            var info = document.getElementById('info');
            if(info){
                var base = info.getAttribute('data-base') || '';
                info.innerHTML = base + ' &nbsp;|&nbsp; 缩放 <span>' + zoomPct + '%</span>';
            }
        };
        _zoomStageEl.addEventListener('wheel', _wheelHandler, { passive: false, capture: true });
    }
    function unbindWheelZoom(){
        if(_zoomStageEl && _wheelHandler){
            // capture:true 蹇呴』涓?addEventListener 鏃朵竴鑷存墠鑳芥垚鍔熺Щ闄?
            _zoomStageEl.removeEventListener('wheel', _wheelHandler, { capture: true });
            _zoomStageEl = null;
            _wheelHandler = null;
        }
    }

    // ---- 榧犳爣鎷栨嫿骞崇Щ ----  //
    function bindDrag(){
        unbindDrag();
        var stage = document.getElementById('zoom-stage');
        if(!stage) return;
        _wheelTarget = stage;   // _wheelTarget 浠呬緵 drag 浣跨敤
        stage.addEventListener('mousedown', _onDragStart);
        stage.addEventListener('click', _onSuppressClick, true); // 鎹曡幏闃舵鎷︽埅
        document.addEventListener('mousemove', _onDragMove);
        document.addEventListener('mouseup', _onDragEnd);
    }
    function unbindDrag(){
        if(_wheelTarget){
            _wheelTarget.removeEventListener('mousedown', _onDragStart);
            _wheelTarget.removeEventListener('click', _onSuppressClick, true);
            _wheelTarget = null;
        }
        document.removeEventListener('mousemove', _onDragMove);
        document.removeEventListener('mouseup', _onDragEnd);
        _dragActive = false;
        _wasDragging = false;
    }
    function _onDragStart(e){
        if(e.button !== 0) return; // 鍙搷搴斿乏閿?
        _dragStartX = e.clientX;
        _dragStartY = e.clientY;
        _dragOrigTX = _translateX;
        _dragOrigTY = _translateY;
        _dragActive = false; // 绛夊緟绉诲姩瑙﹀彂
        _wasDragging = false;
    }
    function _onDragMove(e){
        if(_dragStartX === 0 && _dragStartY === 0) return; // 鏈寜涓?
        var dx = e.clientX - _dragStartX;
        var dy = e.clientY - _dragStartY;
        if(!_dragActive){
            if(Math.abs(dx) < DRAG_THRESHOLD && Math.abs(dy) < DRAG_THRESHOLD) return;
            _dragActive = true;
            _wasDragging = true;
            var stage = document.getElementById('zoom-stage');
            if(stage) stage.classList.add('dragging');
        }
        _translateX = _dragOrigTX + dx;
        _translateY = _dragOrigTY + dy;
        updateStageTransform();

    function patchSpine38Runtime(){
        // 3.8.99 is still experimental; these guards prevent a few null-value crashes.
        if (_spine38 && _spine38.SkeletonBinary) {
            var _origReadString = _spine38.SkeletonBinary.prototype.readString;
            if (_origReadString && !_origReadString._patched3899) {
                var patchedReadString = function(input) {
                    try {
                        return _origReadString.call(this, input);
                    } catch (e) {
                        if (e && e.message && e.message.indexOf('string table') !== -1) {
                            console.warn('[3.8.99 compat] readString fallback:', e.message);
                            return '';
                        }
                        throw e;
                    }
                };
                patchedReadString._patched3899 = true;
                _spine38.SkeletonBinary.prototype.readString = patchedReadString;
                console.log('[3.8] SkeletonBinary.readString patched for 3.8.99');
            }

            if (_spine38.Color && _spine38.Color.rgba8888ToColor && !_spine38.Color.rgba8888ToColor._patched3899) {
                var _origRGBA = _spine38.Color.rgba8888ToColor;
                var patchedRGBA = function(color, value) {
                    if (value === undefined || value === null) {
                        color.r = 1; color.g = 1; color.b = 1; color.a = 1; return;
                    }
                    _origRGBA.call(this, color, value);
                };
                patchedRGBA._patched3899 = true;
                _spine38.Color.rgba8888ToColor = patchedRGBA;
                console.log('[3.8] Color.rgba8888ToColor patched for 3.8.99');

        }

    function fix38MojibakeName(name){
        if(name == null) return name;
        var fullPlus = String.fromCharCode(0xFF0B);
        var fixed = String(name).trim();
        // Fullwidth plus in 3.8.99 binary can arrive as UTF-8 mojibake.
        fixed = fixed.replace(new RegExp('\u00EF\u00BC\u2039', 'g'), fullPlus);
        fixed = fixed.replace(new RegExp('\uFFEF\uFFBC\uFF8B', 'g'), fullPlus);
        fixed = fixed.replace(/\+/g, fullPlus);
        return fixed;
    }

    function find38RegionByNormalizedName(atlas, name){
        if(!atlas || !atlas.regions) return null;
        var target = fix38MojibakeName(name);
        for(var i = 0; i < atlas.regions.length; i++){
            var region = atlas.regions[i];
            if(region && fix38MojibakeName(region.name) === target) return region;
        }
        return null;
    }

    function patch38AtlasLookup(atlas){
        if(!atlas || !atlas.findRegion || atlas._patchedNameFix) return;
        var origFindRegion = atlas.findRegion;
        atlas.findRegion = function(name){
            var region = origFindRegion.call(this, name);
            if(region) return region;

            var fixed = fix38MojibakeName(name);
            if(fixed && fixed !== name){
                region = origFindRegion.call(this, fixed) || origFindRegion.call(this, fixed + ' ');
                if(region){
                    console.warn('[3.8 atlas] region name fixed:', name, '=>', fixed);
                    return region;
                }
            }

            region = find38RegionByNormalizedName(this, name);
            if(region){
                console.warn('[3.8 atlas] region name normalized:', name, '=>', region.name);
                return region;
            }
            return null;
        };
        atlas._patchedNameFix = true;
    }

    function update38Viewport(){
        if(!_38skeleton || !_38canvas || !_spine38) return;
        var minX = -300, minY = -300, maxX = 300, maxY = 300;
        try {
            _38skeleton.updateWorldTransform();
            if(_38skeleton.getBounds && _spine38.Vector2){
                var offset = new _spine38.Vector2();
                var size = new _spine38.Vector2();
                _38skeleton.getBounds(offset, size, []);
                if(isFinite(offset.x) && isFinite(offset.y) && isFinite(size.x) && isFinite(size.y) && size.x > 1 && size.y > 1){
                    minX = offset.x; minY = offset.y; maxX = offset.x + size.x; maxY = offset.y + size.y;
                }
            } else if(_spine38.SkeletonBounds){
                var bounds = new _spine38.SkeletonBounds();
                bounds.update(_38skeleton, true);
                if(isFinite(bounds.minX) && isFinite(bounds.minY) && isFinite(bounds.maxX) && isFinite(bounds.maxY) && bounds.maxX > bounds.minX && bounds.maxY > bounds.minY){
                    minX = bounds.minX; minY = bounds.minY; maxX = bounds.maxX; maxY = bounds.maxY;
                }
            }
        } catch(e) {
            console.warn('[3.8 viewport] bounds fallback:', e.message);
        }

        var width = Math.max(1, maxX - minX);
        var height = Math.max(1, maxY - minY);
        var pad = 0.72;
        _38viewScale = Math.min(_38canvas.width / width, _38canvas.height / height) * pad;
        if(!isFinite(_38viewScale) || _38viewScale <= 0) _38viewScale = 1;

        var centerX = minX + width / 2;
        var centerY = minY + height / 2;
        _38viewX = _38canvas.width / 2 - centerX * _38viewScale;
        _38viewY = _38canvas.height / 2 + centerY * _38viewScale;
        console.log('[3.8 viewport]', {minX:minX, minY:minY, maxX:maxX, maxY:maxY, scale:_38viewScale, x:_38viewX, y:_38viewY});
    }

    function loop38(){

        }
    }

    // ---- 鍒涘缓鏂?player (鏍囧噯璺緞锛?.x / 3.8 json) ----  //
    function createPlayer(fileObj){
        setStatus('[' + fileObj.version + '] 加载 ' + fileObj.name + '...');

        if(fileObj.supportLevel === 'experimental' && fileObj.supportMessage){
            console.warn('[support]', fileObj.supportMessage, fileObj);
        }

        var dir = getFileDir(fileObj);
        if(!dir){
            setStatus('[' + fileObj.version + '] 当前版本暂不支持直接播放', 'err');
            return;
        }
        var opts = {
            atlasUrl: dir + '/' + fileObj.name + '.atlas',
            alpha: true,
            showControls: true,
            premultipliedAlpha: !!fileObj.pma,
            viewport: {
                debugRender: false,
                padLeft: '8%',
                padRight: '8%',
                padTop: '12%',
                padBottom: '12%'
            },
            success: function(player){
                setStatus('[' + fileObj.version + '] 加载完成', 'ok');
                info.innerHTML =
                    '版本 <span>' + fileObj.version + '</span>' +
                    ' &nbsp;|&nbsp; 文件 <span>' + fileObj.name + '</span>';
                var actualVer = getEffectiveVersion(fileObj);
                if(actualVer && actualVer !== fileObj.version)
                    info.innerHTML += ' &nbsp;|&nbsp; 实际版本 <span>' + actualVer + '</span>';
                info.innerHTML += ' &nbsp;|&nbsp; 格式 <span>.' + fileObj.ext + '</span>';
                // 缁戝畾婊氳疆缂╂斁 + 鎷栨嫿
                bindWheelZoom(player);
                bindDrag();
                if(player && player.skeleton && player.skeleton.data){
                    var d = player.skeleton.data;
                    // ---- Track 2 表情轨道选择器 ----
                    var track2Wrap = document.getElementById('track2-wrap');
                    var track2Sel  = document.getElementById('track2-select');
                    if(track2Wrap && track2Sel && d.animations && d.animations.length > 0){
                        track2Sel.innerHTML = '<option value="">-- 无 --</option>';
                        d.animations.forEach(function(anim){
                            var opt = document.createElement('option');
                            opt.value = anim.name;
                            opt.textContent = anim.name;
                            track2Sel.appendChild(opt);
                        });
                        track2Wrap.style.display = '';
                        track2Sel.onchange = function(){
                            var name = this.value;
                            if(!name || !player.animationState){ return; }
                            try {
                                player.animationState.setAnimation(1, name, true);
                                console.log('[track2] setAnimation(1):', name);
                            } catch(e) {
                                console.warn('[track2] 播放失败:', e.message);
                            }
                        };
                    } else if(track2Wrap){ track2Wrap.style.display = 'none'; }
                    if(d.animations && d.animations.length){
                        info.innerHTML += ' &nbsp;|&nbsp; 动画 <span>' + d.animations.length + '</span> 个';
                        // 鑷姩鎾斁锛氬厛 setAnimation锛屽啀 click 鎾斁鎸夐挳鍏滃簳
                        var firstAnim = d.animations[0].name;

                        setTimeout(function(){
                            // 绉婚櫎鏂瑰紡1鐨?return锛岃涓ょ鏂瑰紡閮芥墽琛屽厹搴?
                            try {
                                player.animationState.setAnimation(0, firstAnim, true);
                                console.log('[autoplay] setAnimation done');
                            } catch(e) {}
                            console.log('[autoplay] skip player button click; manual controls remain available');
                            return;
                            // 鏂瑰紡2锛氱洿鎺?click 鎾斁鎸夐挳锛堢敤鎴风‘璁?click 鏈夋晥锛?

                                var btns = document.querySelectorAll('#spine-player button');
                                console.log('[autoplay] found', btns.length, 'buttons');
                                for(var i=0;i<btns.length;i++){
                                    var b = btns[i];
                                    var tip = (b.title||b.textContent||b.getAttribute('aria-label')||'').toLowerCase();
                                    console.log('[autoplay] btn['+i+']:', tip, b.className);
                                    if(/play/i.test(tip) || b.querySelector('svg,path')){
                                        b.click();
                                        console.log('[autoplay] clicked btn['+i+']');
                                        return;
                                    }
                                }
                                // 閮芥病鎵惧埌锛宑lick 绗竴涓寜閽?
                                if(btns.length) { btns[0].click(); console.log('[autoplay] clicked btns[0] fallback'); }
                                else console.warn('[autoplay] no buttons found');
                            } catch(e2) {
                                console.warn('[autoplay] click failed', e2);
                            }
                        }, 1000);
                    }
                    if(d.bones)      info.innerHTML += ' &nbsp;|&nbsp; 骨骼 <span>' + d.bones.length + '</span> 个';
                    if(d.skins)      info.innerHTML += ' &nbsp;|&nbsp; 皮肤 <span>' + d.skins.length + '</span> 个';

            },
            error: function(player, reason){

            }
        };

        if(fileObj.ext === 'json'){
            opts.jsonUrl = dir + '/' + fileObj.name + '.json';
        } else {
            opts.skelUrl = dir + '/' + fileObj.name + '.skel';
        }

            return;
        }
        console.log('[version]', fileObj.version, 'actual:', getEffectiveVersion(fileObj), '-> runtime:', (api === spineRuntime['4.0'] ? '4.0' : (api === spineRuntime['4.2'] ? '4.2' : '4.1')));
        try{
            currentPlayer = new api.SpinePlayer('spine-player', opts);
        }catch(e){
            setStatus('初始化异常: ' + e.message, 'err');
        }
    }

    var _rtLoaded = {};

    // CDN 回退列表: 优先 jsdelivr → 失败则 unpkg
    var _cdnBases = [
        'https://cdn.jsdelivr.net/npm/@esotericsoftware/spine-player@',
        'https://unpkg.com/@esotericsoftware/spine-player@'
    ];

    function loadRuntime(version, cb){
        if(_rtLoaded[version]){ cb(); return; }
        setStatus('加载 Spine ' + version + ' 运行时...');
        var _cdnIdx = 0;
        function _tryLoad(){
            var s = document.createElement('script');
            s.src = _cdnBases[_cdnIdx] + version + '/dist/iife/spine-player.js';
            s.onload = function(){
                spineRuntime[version] = window.spine;
                _rtLoaded[version] = true;
                console.log('[runtime] loaded', version, 'from', _cdnBases[_cdnIdx]);
                cb();
            };
            s.onerror = function(){
                console.warn('[runtime] CDN failed:', _cdnBases[_cdnIdx], 'trying next...');
                _cdnIdx++;
                if(_cdnIdx < _cdnBases.length){
                    _tryLoad();
                } else {
                    setStatus('Spine ' + version + ' 所有 CDN 加载失败', 'err');
                }
            };
            document.head.appendChild(s);
        }
        _tryLoad();
    }

    function waitBothRuntimes(){
        // 并行加载所有版本运行时（原来串行加载，总耗时 = 各版本之和；并行后 ≈ 最慢的那个）
        var versions = ['4.0', '4.1', '4.2'];
        var done = 0;
        var total = versions.length;

        setStatus('加载 Spine 运行时 (0/' + total + ')...');

        versions.forEach(function(ver){
            loadRuntime(ver, function(){
                done++;
                setStatus('加载 Spine 运行时 (' + done + '/' + total + ')...');
                if(done === total){
                    fetchFileList(function(firstFile){
                        switchSpine(0);
                    });
                    fetchBackgrounds();
                }
            });
        });
    }

    // 降级模式检测：file:// 直接打开，或服务器 API 不可用时走 GIF 预览
    function initApp(){
        if(checkDegradedMode()){
            // file:// 协议：无法 fetch 目录，直接使用 build 脚本注入的 _gifFiles
            startDegradedMode(_gifFiles);
            return;
        }
        // http:// 下探测服务器 API 是否可用（比赛平台无后端时自动降级）
        fetch('/api/file-list')
            .then(function(r){ return r.json(); })
            .then(function(list){
                if(list && list.length){
                    // 服务器有 spine 文件，走正常流程
                    waitBothRuntimes();
                } else {
                    // 服务器无 spine 文件，尝试自动扫描 .gif
                    fetch('/api/gif-list')
                        .then(function(r){ return r.json(); })
                        .then(function(gifs){
                            startDegradedMode(gifs);
                        })
                        .catch(function(){
                            startDegradedMode(_gifFiles);
                        });
                }
            })
            .catch(function(){
                // 服务器 API 不可用，降级到 GIF 预览
                startDegradedMode(_gifFiles);
            });
    }

    if(document.readyState === 'loading'){
        document.addEventListener('DOMContentLoaded', initApp);
    } else {
        initApp();
    }
