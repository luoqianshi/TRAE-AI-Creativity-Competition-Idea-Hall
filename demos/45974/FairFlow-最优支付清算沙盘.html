function FairFlow_Settle(matrix) {
    // Step 1: 计算每个节点的净余额
    balances = calculate_net_balance(matrix)
    
    // Step 2: 分离债权人和债务人
    creditors = filter(balances, balance > 0)
    debtors = filter(balances, balance < 0)
    
    // Step 3: 贪心匹配算法
    transactions = []
    while (creditors.length > 0 && debtors.length > 0) {
        max_creditor = find_max(creditors)
        max_debtor = find_min(debtors)
        
        amount = min(abs(max_debtor.balance), max_creditor.balance)
        transactions.push({
            from: max_debtor.name,
            to: max_creditor.name,
            amount: amount
        })
        
        update_balances(max_creditor, max_debtor, amount)
    }
    
    return optimize(transactions)  // 最少交易次数 ≤ N-1
}
